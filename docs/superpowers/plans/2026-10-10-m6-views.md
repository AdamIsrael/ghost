# M6 View Editing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** In the TUI, `n` creates a view, `e` edits the selected one, `d` deletes it (after `y` to confirm). Views edited this way are saved to `views.toml` next to `config.toml`; `config.toml` is never written by Ghost. Changes take effect immediately: the poller and engine switch to the new views and re-poll.

**Architecture:** `config` loads views from both files (tagging each with its `Source`) and gains `SaveViews` (atomic write of `views.toml`). Views defined in `config.toml` stay read-only in the TUI. `poller` and `engine` gain `SetViews`. The TUI owns a mutable view list, a small form component (Bubbles v2 `textinput`), and a delete confirmation; it calls an injected `SaveViews` that `cmd/ghost` implements (validate the full config → write `views.toml` → update poller and engine).

**Tech Stack:** Go 1.27, `charm.land/bubbles/v2` v2.2.1 (`textinput`), existing deps.

**Spec:** `docs/superpowers/specs/2026-10-09-ghost-design.md` (updated in Task 1: views move to `views.toml`).

**Bead:** `ghost-qz7.6`

## Global Constraints

- Module path `github.com/AdamIsrael/ghost`.
- **Operator decision (2026-10-10):** Ghost never writes `config.toml` (it would lose comments). TUI-managed views live in `views.toml` in the same directory.
- Views from `config.toml` keep working and are shown, but `e`/`d` on them only flash "defined in config.toml; edit it there".
- View names are unique across both files; the existing `Validate` enforces this on the merged list.
- `views.toml` is written atomically (temp file + rename) with a header comment; it holds only `[[view]]` tables.
- `ghost init` writes the starter view to `views.toml`; `default_config.toml` no longer contains `[[view]]`.
- Saving runs full config validation first; on error the form stays open and shows the error; nothing is written.
- Keys: `n` new, `e` edit, `d` delete (`y` confirms, anything else cancels). In the form: `tab`/`↓` next field, `shift+tab`/`↑` previous, `space` toggles auto-review, `←`/`→` cycle the agent, `enter` saves, `esc` cancels.
- **Git:** one commit per task on branch `m6-views`; ask before merging or pushing.

## Review Focus

1. Saving never touches `config.toml` (Task 1 `TestSaveViewsLeavesConfigUntouched`, Task 5 `TestTUISaveWritesViewsFileOnly`).
2. A crash mid-save can't leave a half-written `views.toml` (Task 1 atomic rename; `TestSaveViewsAtomicLeavesNoTemp`).
3. Invalid input (empty query, duplicate name, unknown agent) → error in the form, nothing saved (Task 4 `TestFormShowsSaveError`).
4. Deleting the last / selected view leaves a valid selection (Task 4 `TestDeleteViewConfirm`).
5. New views poll immediately; deleted views stop polling and stop auto-review (Task 2 `TestSetViewsPollsNewViews`, `TestEngineSetViews`).

package config

import (
	"bytes"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"strings"

	"github.com/BurntSushi/toml"
)

// ViewsFileName holds the views Ghost manages from the TUI. It sits next to
// config.toml, which Ghost never writes (so its comments survive).
const ViewsFileName = "views.toml"

func ViewsPath(configPath string) string { return filepath.Join(filepath.Dir(configPath), ViewsFileName) }

type viewsFile struct {
	View []View `toml:"view"`
}

const viewsHeader = `# Views managed by Ghost. Edit them in the TUI (n new, e edit, d delete) or here.
# Views can also be defined in config.toml; those are read-only in the TUI.

`

// loadViewsFile reads views.toml; a missing file means no views.
func loadViewsFile(path string) ([]View, error) {
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	var vf viewsFile
	md, err := toml.Decode(string(data), &vf)
	if err != nil {
		return nil, fmt.Errorf("%s: %w", path, err)
	}
	if und := md.Undecoded(); len(und) > 0 {
		keys := make([]string, len(und))
		for i, k := range und {
			keys[i] = k.String()
		}
		return nil, fmt.Errorf("%s: only [[view]] tables belong here; unknown keys: %s", path, strings.Join(keys, ", "))
	}
	for i := range vf.View {
		vf.View[i].Source = ViewsFileName
	}
	return vf.View, nil
}

// SaveViews replaces views.toml with views, atomically: it writes a temp
// file in the same directory and renames it over the old one.
func SaveViews(path string, views []View) error {
	var b bytes.Buffer
	b.WriteString(viewsHeader)
	if len(views) > 0 {
		enc := toml.NewEncoder(&b)
		enc.Indent = ""
		if err := enc.Encode(viewsFile{View: views}); err != nil {
			return err
		}
	}
	tmp, err := os.CreateTemp(filepath.Dir(path), ".views-*.toml")
	if err != nil {
		return err
	}
	defer os.Remove(tmp.Name()) // no-op after a successful rename
	if _, err := tmp.Write(b.Bytes()); err != nil {
		tmp.Close()
		return err
	}
	if err := tmp.Close(); err != nil {
		return err
	}
	if err := os.Chmod(tmp.Name(), 0o644); err != nil {
		return err
	}
	return os.Rename(tmp.Name(), path)
}

	f.query = textinput.New()
	f.query.Placeholder = "is:open is:pr review-requested:@me"
	f.query.SetValue(v.Query)
	for i, a := range f.agents {
		if a == v.Agent {
			f.agentIdx = i
		}
	}
	f.setFocus(fieldName)
	return f
}

func (f *viewForm) setFocus(i int) {
	f.focus = (i + fieldCount) % fieldCount
	f.name.Blur()
	f.query.Blur()
	switch f.focus {
	case fieldName:
		f.name.Focus()
	case fieldQuery:
		f.query.Focus()
	}
}

func (f viewForm) view() config.View {
	return config.View{
		Name:       strings.TrimSpace(f.name.Value()),
		Query:      strings.TrimSpace(f.query.Value()),
		AutoReview: f.auto,
		Agent:      f.agents[f.agentIdx],
	}
}

func (f viewForm) update(msg tea.KeyPressMsg) (viewForm, formAction, tea.Cmd) {
	switch msg.String() {
	case "esc":
		return f, formCancel, nil
	case "enter":
		return f, formSave, nil
	case "tab", "down":
		f.setFocus(f.focus + 1)
		return f, formNone, nil
	case "shift+tab", "up":
		f.setFocus(f.focus - 1)
		return f, formNone, nil
	}
	var cmd tea.Cmd
	switch f.focus {
	case fieldName:
		f.name, cmd = f.name.Update(msg)
	case fieldQuery:
		f.query, cmd = f.query.Update(msg)
	case fieldAuto:
		if msg.String() == "space" {
			f.auto = !f.auto
		}
	case fieldAgent:
		switch msg.String() {
		case "right", "l":
			f.agentIdx = (f.agentIdx + 1) % len(f.agents)
		case "left", "h":
			f.agentIdx = (f.agentIdx - 1 + len(f.agents)) % len(f.agents)
		}
	}
	return f, formNone, cmd
}

func (f viewForm) render(width, height int) string {
	title := "New view"
	if f.editing != "" {
		title = fmt.Sprintf("Edit view %q", f.editing)
	}
	marker := func(i int) string {
		if f.focus == i {
			return "▸ "
		}
		return "  "
	}
	f.name.SetWidth(max(width-16, 10))
	f.query.SetWidth(max(width-16, 10))
	auto := "[ ] off"
	if f.auto {
		auto = "[x] on (reviews cost money)"
	}
	agent := f.agents[f.agentIdx]
	if agent == "" {
		agent = "(default)"
	}
	ls := []string{
		bold.Render("── " + title + " ──"),
		"",
		marker(fieldName) + "Name        " + f.name.View(),
		marker(fieldQuery) + "Query       " + f.query.View(),
		marker(fieldAuto) + "Auto-review " + auto,
		marker(fieldAgent) + "Agent       ← " + agent + " →",
		"",
	}
	if f.err != "" {
		ls = append(ls, errText.Render("✗ "+cleanLine(f.err)), "")
	}
	ls = append(ls, dim.Render("tab next field · space toggle · ←/→ agent · enter save · esc cancel"))
	return block(ls, width, height)
}

func (m Model) handleFormKey(msg tea.KeyPressMsg) (tea.Model, tea.Cmd) {
	f, act, cmd := m.form.update(msg)
	switch act {
	case formCancel:
		m.form = nil
		return m, nil
	case formSave:
		v := f.view()
		v.Source = config.ViewsFileName
		views := append([]config.View(nil), m.views...)
		idx := len(views)
		if f.editing == "" {
			views = append(views, v)
		} else {
			for i := range views {
				if views[i].Name == f.editing {
					views[i], idx = v, i
				}
			}
		}
		if err := m.opts.SaveViews(views); err != nil {
			f.err = err.Error()
			m.form = &f
			return m, nil
		}
		m.views, m.form = views, nil
		m.active, m.cursor = idx, 0
		m.flash = ""
		return m, nil
	}
	m.form = &f
	return m, cmd
}

func (m Model) handleConfirmKey(k string) (tea.Model, tea.Cmd) {
	name := m.confirmDelete
	m.confirmDelete = ""
	if k != "y" {
		return m, nil
	}
	var views []config.View
	for _, v := range m.views {
		if v.Name != name {
			views = append(views, v)
		}
	}
	if err := m.opts.SaveViews(views); err != nil {
		m.flash = "delete: " + err.Error()
		return m, nil
	}
	m.views = views
	if m.active >= len(views) {
		m.active = max(len(views)-1, 0)
	}
	m.cursor = 0
	return m, nil
}
