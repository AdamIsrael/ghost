# M4 Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** While `ghost` runs, PRs in views with `auto_review = true` are reviewed automatically (one review per head commit), within `max_concurrent_reviews` and the daily budget. Finished reviews wait for the operator to press `P` before anything is posted (`post_mode = "confirm"`, the default), or post immediately in `post_mode = "auto"`. Every review is scanned for secrets and blocked from posting if one is found. The TUI shows real spend, running/queued counts, each PR's audit trail and the pending review.

**Architecture:** `store` gains PR/review/audit methods (SQLite, migration 2). `secrets` is a pure regex scanner. `github` gains `PostReview` (REST) with a 422 fallback chain. `engine` owns the queue: it receives poll results, enqueues reviews, runs a worker pool that claims reviews under a budget guard, executes them (GetPR → checkout → agent), scans, and posts on command or automatically. The TUI talks to the engine through a small interface; `cmd/ghost` wires the real engine.

**Tech Stack:** Go 1.27, existing deps. GitHub REST `POST /repos/{owner}/{repo}/pulls/{number}/reviews` via go-gh `RESTClient.DoWithContext`.

**Spec:** `docs/superpowers/specs/2026-10-09-ghost-design.md`

**Bead:** `ghost-qz7.4` (also closes items 1 and 2 of `ghost-qz7.10`)

## Global Constraints

- Module path `github.com/AdamIsrael/ghost`.
- **Operator decisions (2026-10-09):** comments are not posted without confirmation by default → `post_mode = "confirm"` default, `"auto"` opt-in. Every review is secret-scanned before posting; a hit sets state `blocked` and it can never be posted from Ghost.
- One review per (PR, head SHA). A failed review is **not** retried automatically (it may have cost money); `r` re-reviews manually.
- Drafts are never auto-reviewed.
- Budget guard: a review starts only if `spent_today + (running + 1) × max_cost_per_review_usd ≤ max_daily_cost_usd`. Spend is `SUM(cost_usd)` of reviews by `finished_at` in the local day. A run whose cost is unknown counts as `max_cost_per_review_usd`.
- Review states: `queued → running → ready | blocked | failed`, `ready → posted`. On startup, `running` → `queued`.
- Posting: verdict `request_changes` → event `REQUEST_CHANGES`; anything else → `COMMENT` (approvals are M5). Inline comments on side `RIGHT`. On HTTP 422: retry with comments folded into the body; if still 422 and the event was not `COMMENT`, retry once more as `COMMENT` (GitHub rejects REQUEST_CHANGES on your own PR).
- TUI keys added: `r` re-review selected PR, `P` post its pending review, `p` pause/resume the engine.
- Agent/sandbox flags from M3 unchanged.
- **Git:** one commit per task on branch `m4-engine`; ask before merging or pushing.

## Review Focus

1. Two workers claiming at once must not both pass the budget check or claim the same review (Task 5 `TestBudgetGuardStopsNewReviews`, `TestConcurrencyLimit`).
2. A review that fails after spending money is never auto-retried (Task 5 `TestFailedReviewRecordsCostAndIsNotRetried`).
3. A secret in agent output can't be posted, in either post mode (Task 5 `TestSecretInReviewBlocksPosting`, Task 6 `TestAutoModeNeverPostsBlocked`).
4. PR head moves between queueing and running → no stale review (Task 5 `TestSupersededHead`).
5. GitHub rejects inline comments or REQUEST_CHANGES on own PR → still posts, says how (Task 3 `TestPostReviewFallbackChain`).

package store

import (
	"database/sql"
	"errors"
	"time"

	"github.com/AdamIsrael/ghost/internal/github"
)

// Review states.
const (
	Queued  = "queued"
	Running = "running"
	Ready   = "ready"   // finished; waiting to be posted
	Posted  = "posted"
	Blocked = "blocked" // never posted: possible secret in output
	Failed  = "failed"
)

// tsFormat is fixed-width UTC so timestamps sort as strings.
const tsFormat = "2006-01-02T15:04:05.000000000Z"

func ts(t time.Time) string { return t.UTC().Format(tsFormat) }

type Review struct {
	ID         int64
	PRID       int64
	Repo       string
	Number     int
	HeadSHA    string
	Agent      string
	State      string
	Prompt     string
	Verdict    string
	Summary    string
	Comments   string // JSON array of agent.InlineComment
	Error      string
	CostUSD    float64
	CreatedAt  time.Time
	FinishedAt time.Time // set once the review stops running
}

type AuditEvent struct {
	ID       int64
	ReviewID int64 // 0 when not tied to a review
	At       time.Time
	Kind     string
	Message  string
}

func (s *Store) UpsertPR(pr github.PR) (int64, error) {
	var id int64
	err := s.db.QueryRow(`INSERT INTO prs (repo, number, title, author, url, head_sha, status, updated_at)
		VALUES (?, ?, ?, ?, ?, ?, ?, ?)
		ON CONFLICT (repo, number) DO UPDATE SET title = excluded.title, author = excluded.author,
			url = excluded.url, head_sha = excluded.head_sha, status = excluded.status, updated_at = excluded.updated_at
		RETURNING id`,
		pr.Repo, pr.Number, pr.Title, pr.Author, pr.URL, pr.HeadSHA, pr.Status().String(), ts(s.now())).Scan(&id)
	return id, err
}

// EnqueueReview queues a review of headSHA. Unless force is set, it does
// nothing if any review of that head exists, whatever its state — a failed
// review may have cost money, so it is not retried automatically.
func (s *Store) EnqueueReview(prID int64, headSHA, agentName string, force bool) (int64, bool, error) {
	if !force {
		var n int
		if err := s.db.QueryRow(`SELECT COUNT(*) FROM reviews WHERE pr_id = ? AND head_sha = ?`, prID, headSHA).Scan(&n); err != nil {
			return 0, false, err
		}
		if n > 0 {
			return 0, false, nil
		}
	}
	res, err := s.db.Exec(`INSERT INTO reviews (pr_id, head_sha, agent, state, created_at) VALUES (?, ?, ?, ?, ?)`,
		prID, headSHA, agentName, Queued, ts(s.now()))
	if err != nil {
		return 0, false, err
	}
	id, err := res.LastInsertId()
	return id, err == nil, err
}

const reviewCols = `r.id, r.pr_id, p.repo, p.number, r.head_sha, r.agent, r.state, r.prompt, r.verdict,
	r.summary, r.comments, r.error, r.cost_usd, r.created_at, COALESCE(r.finished_at, '')`

func scanReview(row interface{ Scan(...any) error }) (*Review, error) {
	var r Review
	var created, finished string
	err := row.Scan(&r.ID, &r.PRID, &r.Repo, &r.Number, &r.HeadSHA, &r.Agent, &r.State, &r.Prompt, &r.Verdict,
		&r.Summary, &r.Comments, &r.Error, &r.CostUSD, &created, &finished)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	r.CreatedAt, _ = time.Parse(tsFormat, created)
	if finished != "" {
		r.FinishedAt, _ = time.Parse(tsFormat, finished)
	}
	return &r, nil
}

// ClaimNext marks the oldest queued review running and returns it, or nil.
func (s *Store) ClaimNext() (*Review, error) {
	tx, err := s.db.Begin()
	if err != nil {
		return nil, err
	}
	defer tx.Rollback()
	r, err := scanReview(tx.QueryRow(`SELECT `+reviewCols+` FROM reviews r JOIN prs p ON p.id = r.pr_id
		WHERE r.state = ? ORDER BY r.created_at, r.id LIMIT 1`, Queued))
	if err != nil || r == nil {
		return nil, err
	}
	if _, err := tx.Exec(`UPDATE reviews SET state = ? WHERE id = ?`, Running, r.ID); err != nil {
		return nil, err
	}
	r.State = Running
	return r, tx.Commit()
}

// SaveReview writes a review's mutable fields. Leaving the running state
// stamps finished_at once, which is when its cost counts toward the day.
func (s *Store) SaveReview(r *Review) error {
	var finished sql.NullString
	if r.State != Queued && r.State != Running {
		if r.FinishedAt.IsZero() {
			r.FinishedAt = s.now()
		}
		finished = sql.NullString{String: ts(r.FinishedAt), Valid: true}
	}
	if r.Comments == "" {
		r.Comments = "[]"
	}
	_, err := s.db.Exec(`UPDATE reviews SET state = ?, prompt = ?, verdict = ?, summary = ?, comments = ?,
		error = ?, cost_usd = ?, finished_at = ? WHERE id = ?`,
		r.State, r.Prompt, r.Verdict, r.Summary, r.Comments, r.Error, r.CostUSD, finished, r.ID)
	return err
}

// LatestReview returns the most recent review of a PR, or nil.
func (s *Store) LatestReview(repo string, number int) (*Review, error) {
	return scanReview(s.db.QueryRow(`SELECT `+reviewCols+` FROM reviews r JOIN prs p ON p.id = r.pr_id
		WHERE p.repo = ? AND p.number = ? ORDER BY r.id DESC LIMIT 1`, repo, number))
}

func (s *Store) CountState(state string) (int, error) {
	var n int
	err := s.db.QueryRow(`SELECT COUNT(*) FROM reviews WHERE state = ?`, state).Scan(&n)
	return n, err
}

// SpendSince sums the cost of reviews that finished at or after t.
func (s *Store) SpendSince(t time.Time) (float64, error) {
	var sum float64
	err := s.db.QueryRow(`SELECT COALESCE(SUM(cost_usd), 0) FROM reviews WHERE finished_at >= ?`, ts(t)).Scan(&sum)
	return sum, err
}

// ResetRunning requeues reviews left running by a previous process.
func (s *Store) ResetRunning() (int64, error) {
	res, err := s.db.Exec(`UPDATE reviews SET state = ? WHERE state = ?`, Queued, Running)
	if err != nil {
		return 0, err
	}
	return res.RowsAffected()
}

func (s *Store) AddAudit(prID, reviewID int64, kind, msg string) error {
	var rid sql.NullInt64
	if reviewID != 0 {
		rid = sql.NullInt64{Int64: reviewID, Valid: true}
	}
	_, err := s.db.Exec(`INSERT INTO audit_events (pr_id, review_id, kind, message, created_at) VALUES (?, ?, ?, ?, ?)`,
		prID, rid, kind, msg, ts(s.now()))
	return err
}

// Audit returns a PR's last limit events, oldest first.
func (s *Store) Audit(repo string, number, limit int) ([]AuditEvent, error) {
	rows, err := s.db.Query(`SELECT id, review_id, created_at, kind, message FROM (
		SELECT a.id, COALESCE(a.review_id, 0) AS review_id, a.created_at, a.kind, a.message
		FROM audit_events a JOIN prs p ON p.id = a.pr_id
		WHERE p.repo = ? AND p.number = ? ORDER BY a.id DESC LIMIT ?) ORDER BY id`, repo, number, limit)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []AuditEvent
	for rows.Next() {
		var e AuditEvent
		var at string
		if err := rows.Scan(&e.ID, &e.ReviewID, &at, &e.Kind, &e.Message); err != nil {
			return nil, err
		}
		e.At, _ = time.Parse(tsFormat, at)
		out = append(out, e)
	}
	return out, rows.Err()
}

// Package secrets detects credential-shaped strings in text Ghost is about to
// post, as a last line of defence against an agent leaking one.
package secrets

import (
	"regexp"
	"strings"
)

type Finding struct {
	Kind     string
	Redacted string
}

var patterns = []struct {
	kind string
	re   *regexp.Regexp
}{
	{"GitHub token", regexp.MustCompile(`\bgh[pousr]_[A-Za-z0-9]{36,}\b`)},
	{"GitHub fine-grained token", regexp.MustCompile(`\bgithub_pat_[A-Za-z0-9_]{60,}`)},
	{"AWS access key", regexp.MustCompile(`\b(?:AKIA|ASIA)[0-9A-Z]{16}\b`)},
	{"private key", regexp.MustCompile(`-----BEGIN [A-Z ]*PRIVATE KEY-----`)},
	{"Slack token", regexp.MustCompile(`\bxox[abposr]-[A-Za-z0-9-]{10,}`)},
	{"Anthropic API key", regexp.MustCompile(`\bsk-ant-[A-Za-z0-9_-]{20,}`)},
	{"Google API key", regexp.MustCompile(`\bAIza[0-9A-Za-z_-]{35}`)},
}

func Scan(text string) []Finding {
	var out []Finding
	for _, p := range patterns {
		for _, m := range p.re.FindAllString(text, -1) {
			out = append(out, Finding{Kind: p.kind, Redacted: redact(m)})
		}
	}
	return out
}

func redact(s string) string {
	if len(s) <= 8 {
		return "****"
	}
	return s[:6] + "…"
}

// Kinds lists the distinct kinds found, in pattern order.
func Kinds(fs []Finding) string {
	seen := map[string]bool{}
	var kinds []string
	for _, p := range patterns {
		for _, f := range fs {
			if f.Kind == p.kind && !seen[f.Kind] {
				seen[f.Kind] = true
				kinds = append(kinds, f.Kind)
			}
		}
	}
	return strings.Join(kinds, ", ")
}

package github

import (
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"strings"

	"github.com/cli/go-gh/v2/pkg/api"

// REST is the subset of go-gh's RESTClient that Client uses.
type REST interface {
	DoWithContext(ctx context.Context, method, path string, body io.Reader, response interface{}) error
}

func (c *Client) WithREST(r REST) *Client {
	c.rest = r
	return c
}

type ReviewComment struct {
	Path string `json:"path"`
	Line int    `json:"line"`
	Side string `json:"side"`
	Body string `json:"body"`
}

type ReviewPost struct {
	CommitID string          `json:"commit_id"`
	Event    string          `json:"event"` // COMMENT, REQUEST_CHANGES or APPROVE
	Body     string          `json:"body"`
	Comments []ReviewComment `json:"comments,omitempty"`
}

// PostResult says which fallbacks PostReview needed.
type PostResult struct {
	Folded     bool // inline comments moved into the body
	Downgraded bool // event changed to COMMENT
}

// PostReview submits a review. GitHub answers 422 when an inline comment is
// outside the diff, and when you request changes on your own PR, so on 422 it
// retries with the comments folded into the body, then (if the event was not
// already COMMENT) once more as a plain comment.
func (c *Client) PostReview(ctx context.Context, repo string, number int, p ReviewPost) (PostResult, error) {
	path := fmt.Sprintf("repos/%s/pulls/%d/reviews", repo, number)
	for i := range p.Comments {
		if p.Comments[i].Side == "" {
			p.Comments[i].Side = "RIGHT"
		}
	}
	var res PostResult
	err := c.post(ctx, path, p)
	if unprocessable(err) && len(p.Comments) > 0 {
		p.Body += foldComments(p.Comments)
		p.Comments = nil
		res.Folded = true
		err = c.post(ctx, path, p)
	}
	if unprocessable(err) && p.Event != "COMMENT" {
		p.Event = "COMMENT"
		res.Downgraded = true
		err = c.post(ctx, path, p)
	}
	if err != nil {
		return PostResult{}, fmt.Errorf("post review: %w", err)
	}
	return res, nil
}

func (c *Client) post(ctx context.Context, path string, body any) error {
	data, err := json.Marshal(body)
	if err != nil {
		return err
	}
	return c.rest.DoWithContext(ctx, "POST", path, bytes.NewReader(data), nil)
}

func unprocessable(err error) bool {
	var h *api.HTTPError
	return errors.As(err, &h) && h.StatusCode == 422
}

func foldComments(cs []ReviewComment) string {
	var b strings.Builder
	b.WriteString("\n\n**Inline comments** (GitHub could not attach these to the diff):\n")
	for _, c := range cs {
		fmt.Fprintf(&b, "- `%s:%d` %s\n", c.Path, c.Line, strings.ReplaceAll(c.Body, "\n", " "))
	}
	return b.String()
}

// lockRepo serializes work on one repo's bare clone across goroutines (mutex)
// and across ghost processes (flock on <root>/repos/<repo>.lock).
func (m *Manager) lockRepo(repo string) (func(), error) {
	mu := m.repoLock(repo)
	mu.Lock()
	path := filepath.Join(m.root, "repos", repo+".lock")
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		mu.Unlock()
		return nil, err
	}
	f, err := os.OpenFile(path, os.O_CREATE|os.O_RDWR, 0o600)
	if err != nil {
		mu.Unlock()
		return nil, err
	}
	if err := syscall.Flock(int(f.Fd()), syscall.LOCK_EX); err != nil {
		f.Close()
		mu.Unlock()
		return nil, err
	}
	return func() {
		syscall.Flock(int(f.Fd()), syscall.LOCK_UN)
		f.Close()
		mu.Unlock()
	}, nil
}

// Package engine runs reviews: it queues PRs from poll results, runs them
// with a bounded worker pool under the daily budget, and records everything.
package engine

import (
	"context"
	"encoding/json"
	"fmt"
	"strings"
	"sync"
	"time"

	"github.com/AdamIsrael/ghost/internal/agent"
	"github.com/AdamIsrael/ghost/internal/config"
	"github.com/AdamIsrael/ghost/internal/github"
	"github.com/AdamIsrael/ghost/internal/poller"
	"github.com/AdamIsrael/ghost/internal/secrets"
	"github.com/AdamIsrael/ghost/internal/store"
	"github.com/AdamIsrael/ghost/internal/workspace"
)

type Deps struct {
	Store       *store.Store
	GetPR       func(ctx context.Context, repo string, number int) (github.PRDetail, error)
	Checkout    func(ctx context.Context, repo string, number int, baseRef, headSHA string) (workspace.Workspace, func(), error)
	NewReviewer func(name string, a config.Agent) (agent.Reviewer, error)
	PostReview  func(ctx context.Context, repo string, number int, p github.ReviewPost) (github.PostResult, error)
	Now         func() time.Time
}

type Snapshot struct {
	SpendTodayUSD float64
	MaxDailyUSD   float64
	Running       int
	Queued        int
	Paused        bool
	BudgetPaused  bool // reviews are queued but the daily budget is reached
}

type PRState struct {
	Audit    []store.AuditEvent
	Latest   *store.Review
	Comments []agent.InlineComment // Latest's comments, decoded
}

// tick re-checks the queue (budget may have reset at midnight).
const tick = 30 * time.Second

type Engine struct {
	cfg     config.Config
	d       Deps
	polls   chan poller.Result
	wake    chan struct{}
	changes chan struct{}

	mu           sync.Mutex
	paused       bool
	budgetPaused bool
	running      int
}

func New(cfg config.Config, d Deps) *Engine {
	if d.Now == nil {
		d.Now = time.Now
	}
	return &Engine{
		cfg:     cfg,
		d:       d,
		polls:   make(chan poller.Result, 16),
		wake:    make(chan struct{}, 1),
		changes: make(chan struct{}, 1),
	}
}

// Changes signals (coalesced) whenever engine state may have changed.
func (e *Engine) Changes() <-chan struct{} { return e.changes }

func (e *Engine) notify() {
	select {
	case e.changes <- struct{}{}:
	default:
	}
}

func (e *Engine) poke() {
	select {
	case e.wake <- struct{}{}:
	default:
	}
}

// HandlePoll hands a poll result to the engine without blocking. If the
// engine is busy the result is dropped; the next poll carries the same PRs.
func (e *Engine) HandlePoll(r poller.Result) {
	select {
	case e.polls <- r:
	default:
	}
}

// Run requeues reviews interrupted by a previous exit, then processes poll
// results and runs reviews until ctx is cancelled. Reviews still running at
// cancellation stay "running" and are requeued on the next start.
func (e *Engine) Run(ctx context.Context) {
	if n, err := e.d.Store.ResetRunning(); err == nil && n > 0 {
		e.notify()
	}
	for i := 0; i < e.cfg.MaxConcurrentReviews; i++ {
		go e.worker(ctx)
	}
	t := time.NewTicker(tick)
	defer t.Stop()
	e.poke()
	for {
		select {
		case <-ctx.Done():
			return
		case r := <-e.polls:
			e.enqueue(r)
		case <-t.C:
			e.poke()
			e.notify()
		}
	}
}

func (e *Engine) worker(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		case <-e.wake:
		}
		for ctx.Err() == nil {
			r := e.claim()
			if r == nil {
				break
			}
			e.poke() // let another worker claim the next one
			e.process(ctx, r)
		}
	}
}

func (e *Engine) view(name string) (config.View, bool) {
	for _, v := range e.cfg.Views {
		if v.Name == name {
			return v, true
		}
	}
	return config.View{}, false
}

func (e *Engine) enqueue(res poller.Result) {
	v, ok := e.view(res.View)
	if !ok || !v.AutoReview || res.PRs == nil {
		return
	}
	agentName := v.Agent
	if agentName == "" {
		agentName = e.cfg.DefaultAgent
	}
	created := false
	for _, pr := range res.PRs {
		if pr.IsDraft || pr.HeadSHA == "" {
			continue
		}
		prID, err := e.d.Store.UpsertPR(pr)
		if err != nil {
			continue
		}
		id, ok, err := e.d.Store.EnqueueReview(prID, pr.HeadSHA, agentName, false)
		if err != nil || !ok {
			continue
		}
		e.audit(prID, id, "queued", fmt.Sprintf("view %q, agent %s, commit %.7s", v.Name, agentName, pr.HeadSHA))
		created = true
	}
	if created {
		e.poke()
		e.notify()
	}
}

func startOfDay(t time.Time) time.Time {
	return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, t.Location())
}

// claim takes the next queued review if the engine isn't paused and the
// budget allows one more concurrent review at the per-review cap.
func (e *Engine) claim() *store.Review {
	e.mu.Lock()
	defer e.mu.Unlock()
	if e.paused {
		return nil
	}
	spend, err := e.d.Store.SpendSince(startOfDay(e.d.Now()))
	if err != nil {
		return nil
	}
	if spend+float64(e.running+1)*e.cfg.MaxCostPerReviewUSD > e.cfg.MaxDailyCostUSD {
		queued, _ := e.d.Store.CountState(store.Queued)
		e.budgetPaused = queued > 0
		return nil
	}
	r, err := e.d.Store.ClaimNext()
	if err != nil || r == nil {
		return nil
	}
	e.budgetPaused = false
	e.running++
	return r
}

func (e *Engine) process(ctx context.Context, r *store.Review) {
	defer func() {
		e.mu.Lock()
		e.running--
		e.mu.Unlock()
		e.notify()
		e.poke()
	}()
	e.audit(r.PRID, r.ID, "started", "agent "+r.Agent)
	e.notify()
	if err := e.review(ctx, r); err != nil {
		if ctx.Err() != nil {
			return // shutting down: left running, requeued on next start
		}
		r.State, r.Error = store.Failed, err.Error()
		e.save(r)
		e.audit(r.PRID, r.ID, "failed", fmt.Sprintf("%v (spent $%.2f)", err, r.CostUSD))
	}
}

func (e *Engine) review(ctx context.Context, r *store.Review) error {
	a, ok := e.cfg.Agents[r.Agent]
	if !ok {
		return fmt.Errorf("agent %q is no longer configured", r.Agent)
	}
	pr, err := e.d.GetPR(ctx, r.Repo, r.Number)
	if err != nil {
		return err
	}
	if pr.HeadSHA != r.HeadSHA {
		return fmt.Errorf("superseded: PR head moved to %.7s", pr.HeadSHA)
	}
	rendered, err := RenderPrompt(&e.cfg, r.Agent, pr)
	if err != nil {
		return err
	}
	reviewer, err := e.d.NewReviewer(r.Agent, a)
	if err != nil {
		return err
	}
	ws, cleanup, err := e.d.Checkout(ctx, r.Repo, r.Number, pr.BaseRef, pr.HeadSHA)
	if err != nil {
		return err
	}
	defer cleanup()

	r.Prompt = agent.BuildPrompt(rendered, ws.DiffPath, ws.CommitsPath)
	res, err := reviewer.Review(ctx, agent.ReviewRequest{
		Dir: ws.Dir, ContextDir: ws.ContextDir, Prompt: r.Prompt, MaxCostUSD: e.cfg.MaxCostPerReviewUSD,
	})
	r.CostUSD = res.CostUSD
	if err != nil {
		return err
	}
	comments, err := json.Marshal(res.Comments)
	if err != nil {
		return err
	}
	r.Verdict, r.Summary, r.Comments = string(res.Verdict), res.Summary, string(comments)

	if found := secrets.Scan(reviewText(r.Summary, res.Comments)); len(found) > 0 {
		r.State, r.Error = store.Blocked, "possible secret in review: "+secrets.Kinds(found)
		e.save(r)
		e.audit(r.PRID, r.ID, "blocked", r.Error+"; it will not be posted")
		return nil
	}
	r.State = store.Ready
	e.save(r)
	e.audit(r.PRID, r.ID, "reviewed", fmt.Sprintf("%s, %d comments, $%.2f", res.Verdict, len(res.Comments), res.CostUSD))
	e.audit(r.PRID, r.ID, "ready to post", "press P to post it")
	return nil
}

func reviewText(summary string, comments []agent.InlineComment) string {
	var b strings.Builder
	b.WriteString(summary)
	for _, c := range comments {
		b.WriteString("\n" + c.Path + "\n" + c.Body)
	}
	return b.String()
}

func (e *Engine) save(r *store.Review) { _ = e.d.Store.SaveReview(r) }

func (e *Engine) audit(prID, reviewID int64, kind, msg string) {
	_ = e.d.Store.AddAudit(prID, reviewID, kind, msg)
	e.notify()
}

// TogglePause stops or resumes starting new reviews and returns the new state.
func (e *Engine) TogglePause() bool {
	e.mu.Lock()
	e.paused = !e.paused
	paused := e.paused
	e.mu.Unlock()
	if !paused {
		e.poke()
	}
	e.notify()
	return paused
}

func (e *Engine) Snapshot() Snapshot {
	spend, _ := e.d.Store.SpendSince(startOfDay(e.d.Now()))
	queued, _ := e.d.Store.CountState(store.Queued)
	e.mu.Lock()
	defer e.mu.Unlock()
	return Snapshot{
		SpendTodayUSD: spend, MaxDailyUSD: e.cfg.MaxDailyCostUSD,
		Running: e.running, Queued: queued, Paused: e.paused, BudgetPaused: e.budgetPaused,
	}
}

// PRState returns a PR's recent audit trail and latest review.
func (e *Engine) PRState(repo string, number int) PRState {
	var st PRState
	st.Audit, _ = e.d.Store.Audit(repo, number, 20)
	st.Latest, _ = e.d.Store.LatestReview(repo, number)
	if st.Latest != nil {
		_ = json.Unmarshal([]byte(st.Latest.Comments), &st.Comments)
	}
	return st
}

// PostPending posts a PR's finished review (state ready).
func (e *Engine) PostPending(ctx context.Context, repo string, number int) error {
	r, err := e.d.Store.LatestReview(repo, number)
	if err != nil {
		return err
	}
	if r == nil || r.State != store.Ready {
		return errors.New("no review is waiting to be posted")
	}
	return e.post(ctx, r)
}

func (e *Engine) post(ctx context.Context, r *store.Review) error {
	var comments []agent.InlineComment
	if err := json.Unmarshal([]byte(r.Comments), &comments); err != nil {
		return err
	}
	// Re-check: the stored text is what gets posted.
	if found := secrets.Scan(reviewText(r.Summary, comments)); len(found) > 0 {
		r.State, r.Error = store.Blocked, "possible secret in review: "+secrets.Kinds(found)
		e.save(r)
		e.audit(r.PRID, r.ID, "blocked", r.Error+"; it will not be posted")
		return errors.New(r.Error)
	}
	p := github.ReviewPost{
		CommitID: r.HeadSHA,
		Event:    "COMMENT",
		Body:     fmt.Sprintf("%s\n\n<sub>Automated review by Ghost · agent `%s` · commit %.7s</sub>", r.Summary, r.Agent, r.HeadSHA),
	}
	if r.Verdict == string(agent.RequestChanges) {
		p.Event = "REQUEST_CHANGES"
	}
	for _, c := range comments {
		label := "**Nit:** "
		if c.Severity == "blocking" {
			label = "**Blocking:** "
		}
		p.Comments = append(p.Comments, github.ReviewComment{Path: c.Path, Line: c.Line, Body: label + c.Body})
	}
	res, err := e.d.PostReview(ctx, r.Repo, r.Number, p)
	if err != nil {
		e.audit(r.PRID, r.ID, "post failed", err.Error())
		return err
	}
	r.State = store.Posted
	e.save(r)
	msg := fmt.Sprintf("%s with %d inline comments", p.Event, len(p.Comments))
	if res.Folded {
		msg += "; comments folded into the body (GitHub rejected their positions)"
	}
	if res.Downgraded {
		msg += "; posted as a comment (GitHub rejected the review event)"
	}
	e.audit(r.PRID, r.ID, "posted", msg)
	return nil
}

// Rereview queues a fresh review of pr's head, using the PR's last agent (or
// the default), unless one is already queued or running.
func (e *Engine) Rereview(pr github.PR) error {
	latest, err := e.d.Store.LatestReview(pr.Repo, pr.Number)
	if err != nil {
		return err
	}
	if latest != nil && (latest.State == store.Queued || latest.State == store.Running) {
		return errors.New("a review of this PR is already queued or running")
	}
	agentName := e.cfg.DefaultAgent
	if latest != nil {
		if _, ok := e.cfg.Agents[latest.Agent]; ok {
			agentName = latest.Agent
		}
	}
	prID, err := e.d.Store.UpsertPR(pr)
	if err != nil {
		return err
	}
	id, _, err := e.d.Store.EnqueueReview(prID, pr.HeadSHA, agentName, true)
	if err != nil {
		return err
	}
	e.audit(prID, id, "queued", fmt.Sprintf("re-review requested, agent %s, commit %.7s", agentName, pr.HeadSHA))
	e.poke()
	return nil
}
