# M1 Skeleton Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `ghost` binary that loads and validates `config.toml` (agents with per-agent review prompts, views, auto-approve rules), opens/migrates the SQLite database, scaffolds defaults with `ghost init`, and verifies `git`/`gh`/`claude` with `ghost --check`.

**Architecture:** Four small packages under `internal/` (`paths`, `config`, `store`, `preflight`) plus a thin `cmd/ghost` that wires them. `cmd/ghost` exposes a testable `run(args, stdout, stderr, runner) int`. No TUI yet (M2). Deviation from spec's store list: views live only in `config.toml` (single source of truth), and daily spend is `SUM(reviews.cost_usd)` rather than a separate table.

**Tech Stack:** Go 1.27, `github.com/BurntSushi/toml`, `modernc.org/sqlite` (pure Go), stdlib `testing` only.

**Spec:** `docs/superpowers/specs/2026-10-09-ghost-design.md`

**Bead:** `ghost-qz7.1`

## Global Constraints

- Module path: `github.com/AdamIsrael/ghost`
- Config default path: `$XDG_CONFIG_HOME/ghost/config.toml`, else `~/.config/ghost/config.toml`
- DB default path: `$XDG_DATA_HOME/ghost/ghost.db`, else `~/.local/share/ghost/ghost.db`
- Config format: TOML. Agents are `[agents.<name>]`; views are `[[view]]`.
- Each agent sets its review prompt via `prompt` (inline) or `prompt_file` (path; `~` expanded; relative paths resolve against the config file's directory). Neither set → built-in default prompt. Both set → error.
- Prompts are Go `text/template`. Vars: `.Repo .Number .Title .Author .Body .BaseRef .HeadRef .HeadSHA .URL .ChangedFiles`.
- Defaults: `poll_interval = "5m"`, `max_concurrent_reviews = 2`, `max_daily_cost_usd = 10.0`, `max_cost_per_review_usd = 1.5`, `auto_approve.require_confirm = true`.
- Only agent kind in M1: `"claude"`.
- No cgo. Tests use stdlib `testing` only.
- **Git:** repo profile is conservative — do not commit unless the user asks. Each task ends with a checkpoint (`go test ./... && go vet ./...`) instead of a commit.

## Review Focus

1. Config file missing → error tells the user to run `ghost init` (Task 1 `TestLoadMissingFile`).
2. Typo'd config key (e.g. `max_concurent_reviews`) → error names the key, not silently ignored (Task 1 `TestParseRejectsUnknownKeys`).
3. `prompt_file = "~/..."` or relative path → resolves correctly (Task 2 `TestAgentPromptFileResolution`).
4. DB directory does not exist yet on first run → created (Task 3 `TestOpenCreatesParentDir`).
5. `gh` installed but not logged in → `--check` fails with "gh auth login" hint (Task 4 `TestRunGhNotAuthenticated`).

// Package config loads and validates Ghost's TOML configuration.
package config

import (
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"strings"
	"time"

	"github.com/BurntSushi/toml"
)

// Duration is a time.Duration that reads from TOML strings like "5m".
type Duration struct{ time.Duration }

func (d *Duration) UnmarshalText(b []byte) error {
	v, err := time.ParseDuration(string(b))
	if err != nil {
		return fmt.Errorf("invalid duration %q: %w", string(b), err)
	}
	d.Duration = v
	return nil
}

func (d Duration) MarshalText() ([]byte, error) { return []byte(d.String()), nil }

type Config struct {
	PollInterval         Duration         `toml:"poll_interval"`
	MaxConcurrentReviews int              `toml:"max_concurrent_reviews"`
	MaxDailyCostUSD      float64          `toml:"max_daily_cost_usd"`
	MaxCostPerReviewUSD  float64          `toml:"max_cost_per_review_usd"`
	DefaultAgent         string           `toml:"default_agent"`
	Agents               map[string]Agent `toml:"agents"`
	AutoApprove          AutoApprove      `toml:"auto_approve"`
	Views                []View           `toml:"view"`

	// baseDir is the directory of the loaded config file; relative
	// prompt_file paths resolve against it.
	baseDir string
}

type Agent struct {
	Kind       string   `toml:"kind"`
	Model      string   `toml:"model"`
	ExtraArgs  []string `toml:"extra_args"`
	Prompt     string   `toml:"prompt"`
	PromptFile string   `toml:"prompt_file"`
}

type AutoApprove struct {
	Enabled            bool `toml:"enabled"`
	RequireConfirm     bool `toml:"require_confirm"`
	RequireCIGreen     bool `toml:"require_ci_green"`
	NoBlockingComments bool `toml:"no_blocking_comments"`
	NoChangesRequested bool `toml:"no_changes_requested"`
	MaxChangedLines    int  `toml:"max_changed_lines"`
}

type View struct {
	Name       string `toml:"name"`
	Query      string `toml:"query"`
	AutoReview bool   `toml:"auto_review"`
	Agent      string `toml:"agent"`
}

// Default returns the configuration used for any key the file leaves unset.
func Default() Config {
	return Config{
		PollInterval:         Duration{5 * time.Minute},
		MaxConcurrentReviews: 2,
		MaxDailyCostUSD:      10,
		MaxCostPerReviewUSD:  1.5,
		Agents:               map[string]Agent{},
		AutoApprove: AutoApprove{
			RequireConfirm:     true,
			RequireCIGreen:     true,
			NoBlockingComments: true,
			NoChangesRequested: true,
			MaxChangedLines:    400,
		},
	}
}

// Parse decodes TOML over Default(). Unknown keys are an error so typos surface.
func Parse(data []byte) (Config, error) {
	cfg := Default()
	md, err := toml.Decode(string(data), &cfg)
	if err != nil {
		return Config{}, err
	}
	if und := md.Undecoded(); len(und) > 0 {
		keys := make([]string, len(und))
		for i, k := range und {
			keys[i] = k.String()
		}
		return Config{}, fmt.Errorf("unknown config keys: %s", strings.Join(keys, ", "))
	}
	return cfg, nil
}

// Load reads and parses the config file at path.
func Load(path string) (Config, error) {
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return Config{}, fmt.Errorf("config not found at %s: run `ghost init` to create one", path)
	}
	if err != nil {
		return Config{}, err
	}
	cfg, err := Parse(data)
	if err != nil {
		return Config{}, fmt.Errorf("%s: %w", path, err)
	}
	cfg.baseDir = filepath.Dir(path)
	return cfg, nil
}

// Package paths resolves Ghost's default file locations.
package paths

import (
	"os"
	"path/filepath"
	"strings"
)

// ExpandHome replaces a leading "~" or "~/" with the user's home directory.
func ExpandHome(p string) string {
	if p != "~" && !strings.HasPrefix(p, "~/") {
		return p
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return p
	}
	return filepath.Join(home, p[1:])
}

func ConfigFile() string { return xdg("XDG_CONFIG_HOME", ".config", "config.toml") }

func DataFile() string { return xdg("XDG_DATA_HOME", ".local/share", "ghost.db") }

func xdg(env, fallback, file string) string {
	if base := os.Getenv(env); base != "" {
		return filepath.Join(base, "ghost", file)
	}
	return filepath.Join(ExpandHome("~/"+fallback), "ghost", file)
}

	add := func(format string, args ...any) { errs = append(errs, fmt.Errorf(format, args...)) }

	}
	if c.MaxCostPerReviewUSD <= 0 {
		add("max_cost_per_review_usd must be greater than 0")
	}

	switch {
	case len(c.Agents) == 0:
		add("at least one [agents.<name>] section is required")
	case c.DefaultAgent == "" && len(c.Agents) == 1:
		for name := range c.Agents {
			c.DefaultAgent = name
		}
	case c.DefaultAgent == "":
		add("default_agent is required when more than one agent is defined")
	default:
		if _, ok := c.Agents[c.DefaultAgent]; !ok {
			add("default_agent %q is not defined under [agents]", c.DefaultAgent)
		}
	}

	names := make([]string, 0, len(c.Agents))
	for name := range c.Agents {
		names = append(names, name)
	}
	sort.Strings(names)
	for _, name := range names {
		a := c.Agents[name]
		switch {
		case a.Kind == "":
			add("agents.%s: kind is required", name)
		case !KnownAgentKinds[a.Kind]:
			add("agents.%s: unknown kind %q", name, a.Kind)
		}
		if a.Prompt != "" && a.PromptFile != "" {
			add("agents.%s: set prompt or prompt_file, not both", name)
			continue
		}
		text, err := c.AgentPrompt(name)
		if err != nil {
			add("agents.%s: %w", name, err)
			continue
		}
		if _, err := RenderPrompt(text, PromptData{}); err != nil {
			add("agents.%s prompt: %w", name, err)
		}
	}

	seen := map[string]bool{}
	for i, v := range c.Views {
		label := fmt.Sprintf("view %q", v.Name)
		if v.Name == "" {
			label = fmt.Sprintf("view %d", i+1)
			add("%s: name is required", label)
		} else if seen[v.Name] {

		}

			if _, ok := c.Agents[v.Agent]; !ok {
				add("%s: unknown agent %q", label, v.Agent)
			}
		}
	}
	return errors.Join(errs...)

// AgentPrompt returns the raw prompt template text for the named agent:
// inline prompt, else the contents of prompt_file, else DefaultPrompt.
func (c *Config) AgentPrompt(name string) (string, error) {

		return "", fmt.Errorf("unknown agent %q", name)

	case a.Prompt != "":
		return a.Prompt, nil

			return "", fmt.Errorf("prompt_file: %w", err)
		}

	}

// Package store persists PRs, reviews and the audit trail in SQLite.
package store

import (
	"database/sql"
	"fmt"
	"os"
	"path/filepath"

	_ "modernc.org/sqlite"
)

type Store struct {
	db *sql.DB
}

// migrations are applied in order; PRAGMA user_version records how many ran.
// Append only — never edit a migration that has shipped.
var migrations = []string{
	`CREATE TABLE prs (
		id         INTEGER PRIMARY KEY,
		repo       TEXT    NOT NULL,
		number     INTEGER NOT NULL,
		title      TEXT    NOT NULL DEFAULT '',
		author     TEXT    NOT NULL DEFAULT '',
		url        TEXT    NOT NULL DEFAULT '',
		head_sha   TEXT    NOT NULL DEFAULT '',
		status     TEXT    NOT NULL DEFAULT '',
		updated_at TEXT    NOT NULL,
		UNIQUE (repo, number)
	);
	CREATE TABLE reviews (
		id          INTEGER PRIMARY KEY,
		pr_id       INTEGER NOT NULL REFERENCES prs(id),
		head_sha    TEXT    NOT NULL,
		agent       TEXT    NOT NULL,
		state       TEXT    NOT NULL,
		prompt      TEXT    NOT NULL DEFAULT '',
		verdict     TEXT    NOT NULL DEFAULT '',
		summary     TEXT    NOT NULL DEFAULT '',
		cost_usd    REAL    NOT NULL DEFAULT 0,
		created_at  TEXT    NOT NULL,
		finished_at TEXT
	);
	CREATE INDEX reviews_pr_sha ON reviews (pr_id, head_sha);
	CREATE INDEX reviews_created ON reviews (created_at);
	CREATE TABLE audit_events (
		id         INTEGER PRIMARY KEY,
		pr_id      INTEGER NOT NULL REFERENCES prs(id),
		review_id  INTEGER REFERENCES reviews(id),
		kind       TEXT    NOT NULL,
		message    TEXT    NOT NULL DEFAULT '',
		created_at TEXT    NOT NULL
	);
	CREATE INDEX audit_pr ON audit_events (pr_id, created_at);`,
}

// Open opens (creating if needed) the database at path and applies migrations.
func Open(path string) (*Store, error) {
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return nil, err
	}
	dsn := "file:" + path + "?_pragma=foreign_keys(1)&_pragma=journal_mode(WAL)&_pragma=busy_timeout(5000)"
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	s := &Store{db: db}
	if err := s.migrate(); err != nil {
		db.Close()
		return nil, fmt.Errorf("migrate %s: %w", path, err)
	}
	return s, nil
}

func (s *Store) Close() error { return s.db.Close() }

func (s *Store) SchemaVersion() (int, error) {
	var v int
	err := s.db.QueryRow(`PRAGMA user_version`).Scan(&v)
	return v, err
}

func (s *Store) migrate() error {
	v, err := s.SchemaVersion()
	if err != nil {
		return err
	}
	for i := v; i < len(migrations); i++ {
		tx, err := s.db.Begin()
		if err != nil {
			return err
		}
		if _, err := tx.Exec(migrations[i]); err != nil {
			tx.Rollback()
			return fmt.Errorf("migration %d: %w", i+1, err)
		}
		if _, err := tx.Exec(fmt.Sprintf(`PRAGMA user_version = %d`, i+1)); err != nil {
			tx.Rollback()
			return err
		}
		if err := tx.Commit(); err != nil {
			return err
		}
	}
	return nil
}

// Package preflight verifies the external tools Ghost shells out to.
package preflight

import (
	"context"
	"errors"
	"os/exec"
	"strings"
	"time"
)

// Runner executes a command and returns its trimmed combined output.
type Runner func(ctx context.Context, name string, args ...string) (string, error)

func ExecRunner(ctx context.Context, name string, args ...string) (string, error) {
	out, err := exec.CommandContext(ctx, name, args...).CombinedOutput()
	return strings.TrimSpace(string(out)), err
}

type Result struct {
	Name   string
	OK     bool
	Detail string
	Fix    string
}

const checkTimeout = 10 * time.Second

// Run checks git, gh (installed and authenticated) and claude.
func Run(ctx context.Context, run Runner) []Result {
	return []Result{
		version(ctx, run, "git", "install git: https://git-scm.com/downloads"),
		gh(ctx, run),
		version(ctx, run, "claude", "install Claude Code: https://docs.claude.com/en/docs/claude-code"),
	}
}

func AllOK(rs []Result) bool {
	for _, r := range rs {
		if !r.OK {
			return false
		}
	}
	return true
}

func version(ctx context.Context, run Runner, name, fix string) Result {
	ctx, cancel := context.WithTimeout(ctx, checkTimeout)
	defer cancel()
	out, err := run(ctx, name, "--version")
	if err != nil {
		detail := err.Error()
		if errors.Is(err, exec.ErrNotFound) {
			detail = "not found on PATH"
		}
		return Result{Name: name, Detail: detail, Fix: fix}
	}
	return Result{Name: name, OK: true, Detail: firstLine(out)}
}

func gh(ctx context.Context, run Runner) Result {
	r := version(ctx, run, "gh", "install gh: https://cli.github.com")
	if !r.OK {
		return r
	}
	ctx, cancel := context.WithTimeout(ctx, checkTimeout)
	defer cancel()
	if _, err := run(ctx, "gh", "auth", "status"); err != nil {
		return Result{Name: "gh", Detail: "not authenticated", Fix: "run `gh auth login`"}
	}
	return r
}

func firstLine(s string) string {
	line, _, _ := strings.Cut(s, "\n")
	return line
}

package config

import (
	_ "embed"
	"errors"
	"io/fs"
	"os"
	"path/filepath"
)

//go:embed default_config.toml
var DefaultConfigTOML string

// WriteDefaults writes a starter config and prompt beside configPath,
// skipping any file that already exists. It returns the files it created.
func WriteDefaults(configPath string) ([]string, error) {
	dir := filepath.Dir(configPath)
	files := []struct{ path, content string }{
		{configPath, DefaultConfigTOML},
		{filepath.Join(dir, "prompts", "default.md"), DefaultPrompt},
	}
	var created []string
	for _, f := range files {
		if err := os.MkdirAll(filepath.Dir(f.path), 0o755); err != nil {
			return created, err
		}
		fh, err := os.OpenFile(f.path, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
		if errors.Is(err, fs.ErrExist) {
			continue
		}
		if err != nil {
			return created, err
		}
		_, werr := fh.WriteString(f.content)
		if cerr := fh.Close(); werr == nil {
			werr = cerr
		}
		if werr != nil {
			return created, werr
		}
		created = append(created, f.path)
	}
	return created, nil
}

// Command ghost watches GitHub pull requests and runs AI code reviews.
package main

import (
	"context"
	"flag"
	"fmt"
	"io"
	"os"

	"github.com/AdamIsrael/ghost/internal/config"
	"github.com/AdamIsrael/ghost/internal/paths"
	"github.com/AdamIsrael/ghost/internal/preflight"
	"github.com/AdamIsrael/ghost/internal/store"
)

func main() {
	os.Exit(run(os.Args[1:], os.Stdout, os.Stderr, preflight.ExecRunner))
}

func run(args []string, stdout, stderr io.Writer, runner preflight.Runner) int {
	fs := flag.NewFlagSet("ghost", flag.ContinueOnError)
	fs.SetOutput(stderr)
	fs.Usage = func() {
		fmt.Fprintln(stderr, "usage: ghost [flags] [init]")
		fs.PrintDefaults()
	}
	cfgPath := fs.String("config", paths.ConfigFile(), "path to config.toml")
	dbPath := fs.String("db", paths.DataFile(), "path to SQLite database")
	check := fs.Bool("check", false, "verify git, gh and claude, then exit")
	if err := fs.Parse(args); err != nil {
		return 2
	}

	switch {
	case fs.Arg(0) == "init":
		return runInit(*cfgPath, stdout, stderr)
	case fs.NArg() > 0:
		fs.Usage()
		return 2
	case *check:
		return runCheck(*cfgPath, stdout, runner)
	}

	cfg, err := config.Load(*cfgPath)
	if err != nil {
		fmt.Fprintln(stderr, "ghost:", err)
		return 1
	}
	st, err := store.Open(*dbPath)
	if err != nil {
		fmt.Fprintln(stderr, "ghost:", err)
		return 1
	}
	defer st.Close()
	fmt.Fprintf(stdout, "ghost: config OK (%d agents, %d views); database at %s. TUI arrives in M2.\n",
		len(cfg.Agents), len(cfg.Views), *dbPath)
	return 0
}

func runInit(cfgPath string, stdout, stderr io.Writer) int {
	created, err := config.WriteDefaults(cfgPath)
	for _, p := range created {
		fmt.Fprintln(stdout, "created", p)
	}
	if err != nil {
		fmt.Fprintln(stderr, "ghost:", err)
		return 1
	}
	if len(created) == 0 {
		fmt.Fprintln(stdout, "nothing to do: config already exists at", cfgPath)
	}
	return 0
}

func runCheck(cfgPath string, stdout io.Writer, runner preflight.Runner) int {
	results := preflight.Run(context.Background(), runner)
	ok := preflight.AllOK(results)
	for _, r := range results {
		mark := "✓"
		if !r.OK {
			mark = "✗"
		}
		fmt.Fprintf(stdout, "%s %-7s %s\n", mark, r.Name, r.Detail)
		if r.Fix != "" {
			fmt.Fprintf(stdout, "          fix: %s\n", r.Fix)
		}
	}
	if _, err := config.Load(cfgPath); err != nil {
		ok = false
		fmt.Fprintf(stdout, "✗ %-7s %v\n", "config", err)
	} else {
		fmt.Fprintf(stdout, "✓ %-7s %s\n", "config", cfgPath)
	}
	if !ok {
		return 1
	}
	return 0
}
